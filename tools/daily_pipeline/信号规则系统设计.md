## 二、大周期过滤（周线级别）

获取周线数据（日线 5 日聚合）：

```python
def weekly_filter(df_daily):
    """周线 MA20 过滤。返回 True=允许交易，False=全仓禁止。"""

    # 取周线 OHLC（周五收盘最晚那根）
    weekly = df_daily.resample('W-FRI').agg({
        'open': 'first', 'high': 'max', 'low': 'min',
        'close': 'last', 'volume': 'sum'
    })
    weekly['ma20'] = weekly['close'].rolling(20).mean()
    weekly['ma20_slope'] = weekly['ma20'].pct_change(5)  # 5 周斜率

    latest_slope = weekly['ma20_slope'].iloc[-1]
    lookback = weekly['ma20_slope'].iloc[-20:]  # 近 20 周

    if latest_slope < lookback.quantile(0.2):
        # 斜率处于近 20 周最低 20% → 趋势很弱，禁止交易
        return False
    return True
```

**允许交易的条件**：周线 MA20 斜率不处于近 20 周的最低 20% 分位。

> 这个阈值（0.2）是初始值，回测时会通过 walk-forward 优化。

## 三、六因子打分系统

### 3.1 因子计算与归一化

所有因子输出统一映射到 `[-1.0, +1.0]` 区间，方便权重叠加。

#### ① 短期动量 (Momentum)

```python
def momentum_score(df, window=5):
    """ROC(5) → [-1, +1]"""
    roc = df['close'].pct_change(window)
    latest = roc.iloc[-1]

    # 用近 60 天分位做自适应映射
    hist = roc.iloc[-60:-1]
    lower = hist.quantile(0.1)   # 很强下跌
    upper = hist.quantile(0.9)   # 很强上涨

    score = (latest - lower) / (upper - lower) * 2 - 1
    return np.clip(score, -1.0, 1.0)
```

#### ② 中期趋势 (Trend)

```python
def trend_score(df, window=20):
    """MA20 斜率 → [-1, +1]"""
    ma = df['close'].rolling(window).mean()
    slope = ma.diff(5) / ma.shift(5)  # 5 日斜率

    latest = slope.iloc[-1]
    hist = slope.iloc[-60:-1]

    # 斜率为正且强 → +1；斜率为负且强 → -1
    lower = hist.quantile(0.1)
    upper = hist.quantile(0.9)

    if latest > 0:
        score = latest / upper if upper > 0 else 0.5
    else:
        score = latest / abs(lower) if lower < 0 else -0.5
    return np.clip(score, -1.0, 1.0)
```

#### ③ 波动率状态 (Volatility)

```python
def volatility_score(df, window=14):
    """ATR 分位 → [-1, +1]（低波动偏正，高波动偏负）"""
    high_low = df['high'] - df['low']
    high_close = abs(df['high'] - df['close'].shift())
    low_close = abs(df['low'] - df['close'].shift())
    tr = pd.concat([high_low, high_close, low_close], axis=1).max(axis=1)
    atr = tr.rolling(window).mean()

    latest = atr.iloc[-1]
    hist = atr.iloc[-90:-1]
    pct = (latest - hist.min()) / (hist.max() - hist.min())  # 0~1

    # 低波动 → 正分（价格稳定，容易出趋势）
    # 高波动 → 负分（市场神经质，信号不可靠）
    score = 1 - pct * 2  # 高波动= -1，低波动= +1
    return np.clip(score, -1.0, 1.0)
```

#### ④ 量价关系 (Volume-Price)

```python
def volume_price_score(df, window=5):
    """量价配合度 → [-1, +1]"""
    price_change = df['close'].pct_change(window)
    vol_change = df['volume'].pct_change(window)

    latest_p = price_change.iloc[-1]
    latest_v = vol_change.iloc[-1]

    # 放量上涨 → 健康 +1
    if latest_p > 0.02 and latest_v > 0.2:
        return 1.0
    # 放量下跌 → 危险 -1
    if latest_p < -0.02 and latest_v > 0.2:
        return -1.0
    # 缩量调整（价格小跌+量缩）→ 偏多
    if -0.02 < latest_p < 0 and latest_v < -0.1:
        return 0.5
    # 缩量上涨 → 力度存疑
    if latest_p > 0.01 and latest_v < -0.1:
        return 0.2
    # 其他 → 中性
    return 0
```

#### ⑤ RSRS（阻力支撑相对强弱）

```python
def rsrs_score(df, window=18):
    """RSRS 指标 → [-1, +1]"""
    high = df['high'].iloc[-window:]
    low = df['low'].iloc[-window:]

    # OLS: high = alpha + beta * low
    # beta 越高 → 支撑越强
    X = sm.add_constant(low.values)
    model = sm.OLS(high.values, X).fit()
    beta = model.params[1]
    r_squared = model.rsquared

    # RSRS 标准分
    rsrs = beta * r_squared

    # 滚动归一化
    hist_beta = ...  # 存历史 RSRS 值
    hist_rsrs = pd.Series(hist_beta)  # 简化：实际要存
    zscore = (rsrs - hist_rsrs.mean()) / hist_rsrs.std()

    # zscore > 0 → 支撑强（偏多），< 0 → 支撑弱（偏空）
    return np.clip(zscore / 2, -1.0, 1.0)  # zscore 缩到 [-1,1]
```

> 注：RSRS 实现细节在回测时调优，Beta 的计算方式可以简化（不用 OLS 直接用 rolling corr）。

#### ⑥ 比价优势 (Relative Strength vs Benchmark)

```python
def relative_strength_score(df_588000, df_benchmark, window=20):
    """588000 vs 科创50(000688) → [-1, +1]"""
    ret_etf = df_588000['close'].pct_change(window)
    ret_bench = df_benchmark['close'].pct_change(window)

    diff = ret_etf.iloc[-1] - ret_bench.iloc[-1]

    # 超额强 → +1，超额弱 → -1
    hist_diff = ret_etf - ret_bench
    hist_diff = hist_diff.iloc[-60:-1]

    lower = hist_diff.quantile(0.1)
    upper = hist_diff.quantile(0.9)
    score = (diff - lower) / (upper - lower) * 2 - 1
    return np.clip(score, -1.0, 1.0)
```

### 3.2 权重与总分

```python
FACTOR_WEIGHTS = {
    'momentum':    0.25,   # 短期动量
    'trend':       0.25,   # 中期趋势
    'volatility':  0.10,   # 波动率状态
    'volume_price': 0.15,  # 量价配合
    'rsrs':        0.15,   # 阻力支撑
    'relative_strength': 0.10,  # 比价优势
}
# 权重总和 = 1.0

def total_score(factors):
    return sum(f * FACTOR_WEIGHTS[k] for k, f in factors.items())
```

> 初始权重是基于逻辑判断分配的。动量 + 趋势占 50% 是主力。
> 这些权重是**可调的**，回测时做敏感性分析。

## 四、三级决策

### 4.1 决策逻辑

```python
def decision(total_score, current_state, weekly_filter_pass):
    """根据总分 + 当前状态 + 大周期过滤 → 输出信号"""

    if not weekly_filter_pass:
        return '空仓'  # 周线走弱，无条件空仓

    # 总分阈值（初始值，回测优化）
    BUY_THRESHOLD  = 0.3    # > 0.3 看多
    SELL_THRESHOLD = -0.2   # < -0.2 看空
    # 中间地带是观望

    if total_score > BUY_THRESHOLD:
        return '持仓'
    elif total_score < SELL_THRESHOLD:
        return '空仓'
    else:
        return '观望'
```

### 4.2 状态机（避免信号震荡）

```
                  ┌────────────────────────────────────┐
                  │                                    │
                  ▼                                    │
   ┌────────┐ 信号转持仓  ┌────────┐ 信号转观望  ┌────────┐
   │ 空仓   │──────────→ │ 持仓   │──────────→ │ 观望   │
   └────────┘            └────────┘            └────────┘
        ↑                    │                      │
        │                    │ 连续 2 天空仓信号      │
        └────────────────────┴──────────────────────┘
                              ↘ 信号转空仓(需持续2天确认)
```

核心规则：
1. **空仓 → 持仓：** 信号转持仓即买入（不做二次确认，怕踏空）
2. **持仓 → 空仓：** 信号转空仓后，**再确认 1 天**，连续 2 天出空仓信号才卖（防假穿）
3. **持仓 → 观望：** 信号转观望后，**减半仓观察 2 天**
4. **观望 → 任何方向：** 正常切换，无需确认

## 五、仓位映射

```python
def position_size(total_score, current_decision):
    """总分 → 仓位百分比"""
    if current_decision == '空仓':
        return 0.0       # 0%
    elif current_decision == '观望':
        return 0.0       # 观望期间不持仓（保守）
        # 另一种哲学：0~20%
    else:  # 持仓
        # 分数越高仓位越高
        score = min(max(total_score, 0), 1)  # 截断到 [0, 1]
        if score > 0.7:
            return 0.7   # 70% 仓位
        else:
            return 0.3 + score * 0.4  # 30%~70% 线性映射
```

## 六、完整的每日运行逻辑

```python
def daily_run(df_daily_588000, df_daily_benchmark):
    """每日运行的唯一入口"""

    # 1. 大周期过滤
    can_trade = weekly_filter(df_daily_588000)

    # 2. 计算各因子分数
    factors = {
        'momentum':    momentum_score(df_daily_588000),
        'trend':       trend_score(df_daily_588000),
        'volatility':  volatility_score(df_daily_588000),
        'volume_price': volume_price_score(df_daily_588000),
        'rsrs':        rsrs_score(df_daily_588000),
        'relative_strength': relative_strength_score(df_daily_588000, df_benchmark),
    }

    # 3. 总分
    score = total_score(factors)

    # 4. 决策
    current_state = get_current_state()  # 从缓存读取
    decision = decide(score, current_state, can_trade)

    # 5. 仓位
    pos = position_size(score, decision)

    # 6. 更新状态缓存
    save_state(decision)

    return {
        'date': today,
        'weekly_filter': can_trade,
        'factors': factors,
        'total_score': score,
        'decision': decision,
        'position': f'{pos*100:.0f}%',
    }
```
